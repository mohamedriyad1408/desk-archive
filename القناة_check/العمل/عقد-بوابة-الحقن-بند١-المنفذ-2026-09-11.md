## ثانيًا — القرار التصميمي (منحة المنفذ — ثلاث طبقات دفاع لا واحدة)

1. **طبقة التطبيق (قائمة، بلا تغيير):** حسابات `Decimal` الدقيقة في `post_financial` وفاحص القيد — لا تُمَسّ.
2. **طبقة المخزن (الجديدة — جوهر البند):** **ختم القيد** + **تسعة مشغّلات** تُثبّت: الصفرية عند الختم، وحصانة التعديل، وحصانة الحذف (بالغًا حالة «بلا أبناء»)، ومنع أي سطر جديد على قيد مختوم.
3. **طبقة الكسر (بوابة الحقن):** اختبارات تفتح اتصالًا مباشرًا وتحاول الكسر بجُمل SQL خالصة — وكل محاولة تُقاس وتُسجَّل.

**آلية الصفرية في المخزن:** لا تقبل SQLite قيود `CHECK` جامعةً عبر الصفوف، فالطريق: عمود `is_sealed` على رأس القيد + مشغّل ختم **يرفض** ختم قيد غير متزن أو بلا أسطر. فيصير: **لا يدخل قيدٌ صالحٌ إلى أي قراءة إلا متزنًا** — والقيد غير المكتمل لا يُقرأ أصلًا (تُرشَّح القراءات بـ`is_sealed = 1`). والدقة المالية تبقى `Decimal` كما هي؛ والفحص المخزني **حساب وحدات ميكرو بعددي صحيح** (لا عائم — رابعًا).

## ثالثًا — النص المخزني الملزم (يُنشأ في الترحيل ١٫٩٫٠)

```sql
-- (١) عمودا الختم والزمن على الرأس، وزمن العمل على السطر (البند ٢)
ALTER TABLE journal_entries  ADD COLUMN is_sealed INTEGER NOT NULL DEFAULT 0;
ALTER TABLE journal_lines    ADD COLUMN business_date TEXT NOT NULL DEFAULT '';
-- ترحيل القيم: business_date للسطور من رأسها؛ وis_sealed=1 لكل رأس متزن قائم.

CREATE TRIGGER IF NOT EXISTS trg_jentry_update_guard BEFORE UPDATE ON journal_entries
WHEN NOT (OLD.is_sealed = 0 AND NEW.is_sealed = 1
    AND OLD.id = NEW.id AND OLD.tenant_id = NEW.tenant_id AND OLD.event_id = NEW.event_id
    AND OLD.business_date = NEW.business_date AND OLD.contract = NEW.contract
    AND OLD.rounding_diff = NEW.rounding_diff AND OLD.posted_by = NEW.posted_by
    AND OLD.created_at = NEW.created_at)
BEGIN SELECT RAISE(ABORT, 'القيد غير قابل للتعديل'); END;
-- (الانتقال الوحيد المسموح: الختم 0→1 وبقية الحقول ثابتة حرفيًا)

CREATE TRIGGER IF NOT EXISTS trg_jentry_seal_requires_balance BEFORE UPDATE ON journal_entries
WHEN (OLD.is_sealed = 0 AND NEW.is_sealed = 1)
  AND ((SELECT COUNT(*) FROM journal_lines l WHERE l.journal_entry_id = NEW.id) = 0
       OR <مجموع(مدين−دائن) بالميكرو للأسطر> <> 0
       OR <سطر غير مرقّم أو سالب أو غير رقمي>)
BEGIN SELECT RAISE(ABORT, 'ختم مرفوض: قيد غير متزن أو بلا أسطر'); END;

CREATE TRIGGER IF NOT EXISTS trg_jlines_no_update BEFORE UPDATE ON journal_lines
BEGIN SELECT RAISE(ABORT, 'سطور القيد غير قابلة للتعديل'); END;

CREATE TRIGGER IF NOT EXISTS trg_jlines_no_delete BEFORE DELETE ON journal_lines
BEGIN SELECT RAISE(ABORT, 'سطور القيد غير قابلة للحذف'); END;

CREATE TRIGGER IF NOT EXISTS trg_jentry_no_delete BEFORE DELETE ON journal_entries
BEGIN SELECT RAISE(ABORT, 'القيد غير قابل للحذف'); END;

CREATE TRIGGER IF NOT EXISTS trg_jlines_no_insert_on_sealed BEFORE INSERT ON journal_lines
WHEN (SELECT is_sealed FROM journal_entries WHERE id = NEW.journal_entry_id) = 1
BEGIN SELECT RAISE(ABORT, 'لا أسطر جديدة على قيد مختوم'); END;

CREATE TRIGGER IF NOT EXISTS trg_ledger_no_update BEFORE UPDATE ON ledger_entries
BEGIN SELECT RAISE(ABORT, 'قيد الدفتر غير قابل للتعديل'); END;

CREATE TRIGGER IF NOT EXISTS trg_ledger_no_delete BEFORE DELETE ON ledger_entries
BEGIN SELECT RAISE(ABORT, 'قيد الدفتر غير قابل للحذف'); END;
```
*(موضعان مختصران: `<مجموع الميكرو>` و`<حارس السطر>` — نصّهما الحرفي في الملف التنفيذي عند أول جولة؛ وأصلهما المُجرَّب: تحويل نص المبلغ إلى ميكرو عبر `substr/instr` بلا `CAST AS REAL`، وحارس السطر بـ`GLOB '*[^0-9.]*'` للمبالغ غير الرقمية — ٤ × ٩ حالات في رابعًا.)*
